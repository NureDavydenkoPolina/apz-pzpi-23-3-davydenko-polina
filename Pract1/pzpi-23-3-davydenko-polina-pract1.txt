В.2 Реальний приклад патерну Iterator
1 class ProfileIterator {
2   constructor(profiles) {
3     this.profiles = profiles;
4     this.position = 0;
5   }
6 
7   hasNext() {
8     return this.position < this.profiles.length;
9   }
10 
11   getNext() {
12     return this.profiles[this.position++];
13   }
14 }
15 
16 class Facebook {
17   constructor() {
18     this.profiles = ["Anna", "John", "Mike"];
19   }
20 
21   createIterator() {
22     return new ProfileIterator(this.profiles);
23   }
24 }
25 
26 const fb = new Facebook();
27 const iterator = fb.createIterator();
28 
29 while (iterator.hasNext()) {
30   console.log(iterator.getNext());
